```javascript
async function renderTeamSection() {
  const container = document.getElementById('section-team');
  if (!container) return;
  container.innerHTML = '<p style="color:#aaa">טוען...</p>';

  try {
    const res  = await fetch('/api/users/group');
    const data = await res.json();
    if (!data.success) throw new Error(data.error);

    const users = data.users;

    container.innerHTML = `
      <div class="section-header" style="display:flex;justify-content:space-between;align-items:center;margin-bottom:16px">
        <h2 style="margin:0">הצוות שלי</h2>
        <button class="btn btn-primary" onclick="showInviteTeamMemberModal()">+ הזמן חבר צוות</button>
      </div>
      ${users.length === 0
        ? '<p style="color:#aaa">אין חברי צוות עדיין.</p>'
        : `<div class="team-list">
            ${users.map(u => renderTeamUserRow(u)).join('')}
           </div>`
      }
    `;
  } catch (err) {
    container.innerHTML = `<p style="color:#e55">שגיאה: ${err.message}</p>`;
  }
}
```

```javascript
function renderGroupUserPermissions(user) {

  return `<div class="permissions-grid" style="display:grid;grid-template-columns:1fr 1fr;gap:8px">
```

**Step 6 — Add savePermissions(userId) function:**
```javascript
async function savePermissions(userId) {
  const checkboxes = document.querySelectorAll(`[data-user="${userId}"][data-perm]`);
  const permissions = {};
  checkboxes.forEach(cb => { permissions[cb.dataset.perm] = cb.checked; });

  try {
    const res  = await fetch(`/api/users/${userId}/permissions`, {
      method:  'PUT',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ permissions }),
    });
    const data = await res.json();
    if (!data.success) throw new Error(data.error);
    showToast('הרשאות עודכנו בהצלחה');
  } catch (err) {
    showToast(`שגיאה: ${err.message}`, 'error');
  }
}
```

```javascript
function promptRemoveGroupUser(userId, userName) {
  // Use a simple confirm dialog first, then ask about products
  // Follow the existing modal pattern in app.js if one exists; otherwise use confirm()
  const keep = confirm(`האם לשמור את המוצרים של ${userName}?\n\nלחץ אישור = שמור מוצרים\nלחץ ביטול = מחק מוצרים`);
  if (!confirm(`האם למחוק את ${userName} מהצוות?`)) return;
  removeGroupUser(userId, keep);
}

async function removeGroupUser(userId, keepProducts) {
  try {
    const res  = await fetch(`/api/users/${userId}?keepProducts=${keepProducts}`, { method: 'DELETE' });
    const data = await res.json();
    if (!data.success) throw new Error(data.error);
    showToast('חבר הצוות הוסר בהצלחה');
    renderTeamSection();
  } catch (err) {
    showToast(`שגיאה: ${err.message}`, 'error');
  }
}
```

**Step 8 — Add showInviteTeamMemberModal() function:**

```javascript
async function showInviteTeamMemberModal() {
  const email = prompt('כתובת אימייל של חבר הצוות החדש:');
  if (!email) return;

  try {
    const res  = await fetch('/api/users/invites', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ email, invitedRole: 'group_user' }),
    });
    const data = await res.json();
    if (!data.success) throw new Error(data.error);
    showToast(`הזמנה נשלחה ל-${email}`);
    if (data.invitation?.inviteUrl) {
      prompt('קישור הזמנה (העתק ושלח ידנית):', data.invitation.inviteUrl);
    }
  } catch (err) {
    showToast(`שגיאה: ${err.message}`, 'error');
  }
}
```
