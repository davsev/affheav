**2. Update createUser() to accept explicit role and groupAdminId:**
```javascript
async function createUser({ googleId, email, name, photo, role, groupAdminId }) {
  const adminEmail = process.env.ADMIN_GOOGLE_EMAIL;
  // Bootstrap: if no explicit role provided and email matches admin, assign 'admin'
  const assignedRole = role || (adminEmail && email === adminEmail ? 'admin' : 'group_user');
  const assignedGroupAdminId = groupAdminId || null;

  const { rows } = await query(
    `INSERT INTO users (google_id, email, name, photo, role, group_admin_id, status)
     VALUES ($1, $2, $3, $4, $5, $6, $7)
     ON CONFLICT (google_id) DO UPDATE
       SET email = EXCLUDED.email,
           name  = EXCLUDED.name,
           photo = EXCLUDED.photo,
           updated_at = NOW()
     RETURNING *`,
    [googleId, email, name, photo, assignedRole, assignedGroupAdminId, 'active']
  );
  const user = _row(rows[0]);
  _cacheSet(googleId, user);
  return user;
}
```
Note: ON CONFLICT intentionally does NOT update role or group_admin_id — existing users keep their role on re-login.

**3. Update updateUserById() to also allow updating permissions and group_admin_id:**

Also update the SQL key mapping — for JSONB, permissions must be cast. Replace the dynamic query builder with one that handles JSONB:
```javascript
async function updateUserById(id, fields) {
  const allowedKeys = ['name', 'photo', 'role', 'status', 'permissions', 'group_admin_id'];
  const updates = [];
  const values  = [];
  let   i       = 1;

  for (const key of allowedKeys) {
    if (fields[key] !== undefined) {
      if (key === 'permissions') {
        updates.push(`permissions = $${i++}::jsonb`);
        values.push(JSON.stringify(fields[key]));
      } else {
        updates.push(`${key} = $${i++}`);
        values.push(fields[key]);
      }
    }
  }
  if (updates.length === 0) return findUserById(id);

  updates.push(`updated_at = NOW()`);
  values.push(id);

  const { rows } = await query(
    `UPDATE users SET ${updates.join(', ')} WHERE id = $${i} RETURNING *`,
    values
  );
  const user = _row(rows[0]);
  if (user?.googleId) _cacheInvalidate(user.googleId);
  return user;
}
```
