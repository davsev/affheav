**middleware/auth.js** — Canonical role guard middleware. The existing local `isAdmin` in server.js checks `role === 'admin'`. Keep that same check for `isSuperAdmin` to avoid breaking existing sessions that still have role='admin' in the 60s cache window.

```javascript
'use strict';

/**
 * isSuperAdmin — allows only role='admin' (SuperAdmin).
 * NOTE: We keep 'admin' as the DB value (not 'super_admin') per research recommendation
 * to avoid breaking existing authenticated sessions.
 */
function isSuperAdmin(req, res, next) {
  if (req.user?.role === 'admin') return next();
  res.status(403).json({ success: false, error: 'Forbidden' });
}

/**
 * isGroupAdminOrAbove — allows 'admin' (SuperAdmin) or 'group_admin'.
 */
function isGroupAdminOrAbove(req, res, next) {
  const role = req.user?.role;
  if (role === 'admin' || role === 'group_admin') return next();
  res.status(403).json({ success: false, error: 'Forbidden' });
}

/**
 * requirePermission(perm) — returns middleware that:
 *   - Passes SuperAdmin and Group Admin unconditionally.
 *   - For group_user: checks req.user.permissions[perm] === true.
 *
 * @param {string} perm - One of: add_products, edit_products, delete_products,
 *   view_logs, trigger_send, manage_schedules, view_settings
 */
function requirePermission(perm) {
  return (req, res, next) => {
    const { role, permissions } = req.user || {};
    if (role === 'admin' || role === 'group_admin') return next();
    if (permissions && permissions[perm] === true) return next();
    res.status(403).json({ success: false, error: 'Forbidden' });
  };
}

module.exports = { isSuperAdmin, isGroupAdminOrAbove, requirePermission };
```

**middleware/scopeGroup.js** — Resolves allowed user IDs for data isolation:

```javascript
'use strict';
const { query } = require('../db');

/**
 * scopeGroup middleware — attaches req.scopedUserIds to the request.
 *
 * SuperAdmin (role='admin'): req.scopedUserIds = null (no filter, sees all)
 * Group Admin (role='group_admin'): req.scopedUserIds = [self.id, ...memberIds]
 * Group User (role='group_user'): req.scopedUserIds = [self.id]
 *
 * Routes use: scopedUserIds === null ? '' : 'AND user_id = ANY($N::uuid[])'
 */
async function scopeGroup(req, res, next) {
  try {
    const { id, role } = req.user;
    if (role === 'admin') {
      req.scopedUserIds = null;
      return next();
    }
    if (role === 'group_admin') {
      const { rows } = await query(
        `SELECT id FROM users WHERE id = $1 OR group_admin_id = $1`,
        [id]
      );
      req.scopedUserIds = rows.map(r => r.id);
      return next();
    }
    // group_user — own data only
    req.scopedUserIds = [id];
    next();
  } catch (err) {
    next(err);
  }
}

module.exports = { scopeGroup };
```
