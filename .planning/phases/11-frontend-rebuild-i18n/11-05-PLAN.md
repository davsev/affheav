**useSseLog.ts:**
```typescript
export interface LogEntry {
  timestamp: string;
  level: string;
  message: string;
}

export function useSseLog(subjectId: string | null) {
  const [entries, setEntries] = useState<LogEntry[]>([]);
  const esRef = useRef<EventSource | null>(null);

  useEffect(() => {
    esRef.current?.close();
    const url = subjectId
      ? `/api/logs?subjectId=${encodeURIComponent(subjectId)}`
      : '/api/logs';
    const es = new EventSource(url);
    es.onmessage = (e) => {
      try {
        const entry = JSON.parse(e.data) as LogEntry;
        setEntries((prev) => [...prev.slice(-499), entry]); // keep last 500
      } catch { /* malformed — skip */ }
    };
    esRef.current = es;
    return () => { es.close(); esRef.current = null; };
  }, [subjectId]);

  return { entries, clear: () => setEntries([]) };
}
```
