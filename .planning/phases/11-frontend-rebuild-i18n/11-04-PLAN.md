**apps/web/src/hooks/useAuth.ts:**
Wraps `useAuthStore`. Decodes the access token (JWT payload via base64 JSON.parse — do NOT import a JWT library) to extract `sub`, `role`, `email`. Exposes:
- `user: { sub, role, email } | null` — null if not authenticated
- `isAdmin: boolean` — `user?.role === 'admin'`
- `accessToken: string | null` — raw token from store
- `logout: () => Promise<void>` — calls `DELETE /api/v1/auth/logout`, clears store, redirects to `/auth/login`

```typescript
function decodeToken(token: string): { sub: string; role: string; email: string } | null {
  try {
    const payload = JSON.parse(atob(token.split('.')[1]));
    return { sub: payload.sub, role: payload.role, email: payload.email };
  } catch {
    return null;
  }
}
```

**apps/web/src/hooks/useLangSwitch.ts:**
```typescript
import { useDirection } from '@mantine/core';
import { useTranslation } from 'react-i18next';
import { api } from '../lib/api';

export function useLangSwitch() {
  const { setDirection } = useDirection();
  const { i18n } = useTranslation();

  async function switchLang(lang: 'he' | 'en') {
    await i18n.changeLanguage(lang);
    setDirection(lang === 'he' ? 'rtl' : 'ltr');
    document.documentElement.lang = lang;
    document.documentElement.dir = lang === 'he' ? 'rtl' : 'ltr';
    // Persist preference — best effort, no throw on failure
    api.patch('/api/users/me/lang', { lang }).catch(() => {});
  }

  return { switchLang, currentLang: i18n.language as 'he' | 'en' };
}
```

```tsx
type TabId = 'dashboard' | 'products' | 'schedules' | 'scraper' | 'add-product'
           | 'aliexpress-search' | 'discover' | 'analytics' | 'logs' | 'settings'
           | 'credentials' | 'users' | 'feature-flags' | 'pending-approvals' | 'my-team';

export function App() {
  const [ready, setReady] = useState(false);
  const [activeTab, setActiveTab] = useState<TabId>('dashboard');
  const setToken = useAuthStore((s) => s.setAccessToken);

      {/* ... render all other pages based on activeTab */}

  );
```
